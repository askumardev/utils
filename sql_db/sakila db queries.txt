1	Display all records from the film table.
SELECT * FROM sakila.film;

2	Display only film title, release year and rental rate.
SELECT title, release_year, rental_rate FROM sakila.film;


3	Find all films released in 2006.
SELECT * FROM sakila.film
where release_year = '2006';


4	Find films with a rental rate greater than 3.
SELECT * FROM sakila.film
where rental_rate > 3;

5	Find films with a rental rate between 1 and 3.
SELECT * FROM sakila.film
where rental_rate between 1 AND 3;


6	Find films with a replacement cost greater than 20.
SELECT * FROM sakila.film
where replacement_cost > 20;


7	Display customers whose first name is 'MARY'.
SELECT * FROM sakila.customer
where first_name like '%mary%';

8	Find customers whose first name starts with 'A'.
SELECT * FROM sakila.customer
where first_name like 'A%';


9	Find films whose title contains 'LOVE'.
SELECT * FROM sakila.film
where title like '%love%';


10	Display films ordered by rental rate descending.
SELECT * FROM sakila.film
order by rental_rate desc;


11	Find the 10 most expensive films based on replacement cost.
SELECT * FROM sakila.film
order by replacement_cost desc
limit 10;


12	Display unique film ratings.
SELECT DISTINCT rating
FROM film;


13	Find films with a rating of 'PG-13'.
SELECT * FROM sakila.film
where rating = 'PG-13';


14	Find customers whose email is NULL.
SELECT * FROM sakila.customer
where email is null;


15	Find films with a rental duration greater than or equal to 6 days.
SELECT * FROM sakila.film
where rental_duration >= 6;
